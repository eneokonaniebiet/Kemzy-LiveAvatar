cmake_minimum_required(VERSION 3.22.1)
project(kemzy_ort_customops LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_HEADER_URL "https://raw.githubusercontent.com/microsoft/onnxruntime/v1.22.0/include/onnxruntime/core/session/onnxruntime_c_api.h")
set(ORT_HEADER "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime_c_api.h")
if(NOT EXISTS "${ORT_HEADER}")
    file(DOWNLOAD "${ORT_HEADER_URL}" "${ORT_HEADER}" TLS_VERIFY ON STATUS DOWNLOAD_STATUS)
    list(GET DOWNLOAD_STATUS 0 DOWNLOAD_CODE)
    if(NOT DOWNLOAD_CODE EQUAL 0)
        message(FATAL_ERROR "Unable to obtain ONNX Runtime C API header: ${DOWNLOAD_STATUS}")
    endif()
endif()
add_library(kemzy_ort_customops SHARED kemzy_ort_customops.cpp)
target_include_directories(kemzy_ort_customops PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(kemzy_ort_customops PRIVATE ORT_API_MANUAL_INIT=1)
target_link_options(kemzy_ort_customops PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
